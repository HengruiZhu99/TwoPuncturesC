cmake_minimum_required(VERSION 3.20)
project(Punctures LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(PUNCTURES_KOKKOS "Build the explicit Kokkos execution backend" OFF)
set(PUNCTURES_KOKKOS_SOURCE "" CACHE PATH "Existing, pinned Kokkos source checkout")
set(HISPID_ROW_POWER 6 CACHE STRING "Positive residual row power: 3 or 6")
option(PUNCTURES_BENCHMARK "Add fenced/timing-only BY Newton instrumentation" OFF)
find_package(GSL REQUIRED)
file(GLOB TP_C CONFIGURE_DEPENDS src/*.c)
list(FILTER TP_C EXCLUDE REGEX "TwoPuncturesRun.c$")
if(PUNCTURES_BENCHMARK)
  list(FILTER TP_C EXCLUDE REGEX "TP_Newton.c$")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/TP_Newton.c validation/by_solver_timer.c)
  file(READ src/TP_Newton.c NEWTON_SOURCE)
  string(REPLACE "%10.3e" "%.17e" NEWTON_SOURCE "${NEWTON_SOURCE}")
  string(REPLACE "|F|=%e" "|F|=%.17e" NEWTON_SOURCE "${NEWTON_SOURCE}")
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/precision_newton.c "${NEWTON_SOURCE}")
  file(READ validation/by_solver_timer.c TIMER_SOURCE)
  string(REPLACE "\"../src/TP_Newton.c\"" "\"${CMAKE_CURRENT_BINARY_DIR}/precision_newton.c\"" TIMER_SOURCE "${TIMER_SOURCE}")
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/by_solver_timer.c "${TIMER_SOURCE}")
  list(APPEND TP_C ${CMAKE_CURRENT_BINARY_DIR}/by_solver_timer.c)
endif()
add_library(TwoPunctures SHARED ${TP_C})
target_include_directories(TwoPunctures PUBLIC include)
target_link_libraries(TwoPunctures PUBLIC GSL::gsl GSL::gslcblas m)
add_library(HiSpID SHARED src/HiSpID_geometry.cpp src/HiSpID_solver.cpp)
target_include_directories(HiSpID PUBLIC include)
target_link_libraries(HiSpID PRIVATE TwoPunctures)
target_compile_definitions(HiSpID PRIVATE HISPID_ROW_POWER=${HISPID_ROW_POWER})
if(PUNCTURES_KOKKOS)
  set(BUILD_SHARED_LIBS ON CACHE BOOL "One shared Kokkos runtime for both puncture DSOs" FORCE)
  if(PUNCTURES_KOKKOS_SOURCE)
    add_subdirectory(${PUNCTURES_KOKKOS_SOURCE} kokkos)
  else()
    find_package(Kokkos 4.7 REQUIRED)
    get_target_property(KOKKOS_RUNTIME_TYPE Kokkos::kokkoscore TYPE)
    if(NOT KOKKOS_RUNTIME_TYPE STREQUAL "SHARED_LIBRARY")
      message(FATAL_ERROR "A shared Kokkos runtime is required for the two puncture DSOs; use a shared package or PUNCTURES_KOKKOS_SOURCE")
    endif()
  endif()
  target_sources(TwoPunctures PRIVATE src/PunctureKokkos.cpp)
  # Keep the HiSpID map builder in the same target as its sampling/maps;
  # target-specific stretch definitions must not drift across the two DSOs.
  target_sources(HiSpID PRIVATE src/HiSpID_setup_kokkos.cpp)
  foreach(t TwoPunctures HiSpID)
    target_compile_definitions(${t} PRIVATE PUNCTURES_KOKKOS=1)
    target_link_libraries(${t} PRIVATE Kokkos::kokkos)
  endforeach()
  add_executable(test_kokkos tests/test_kokkos.cpp)
  target_include_directories(test_kokkos PRIVATE src include)
  target_link_libraries(test_kokkos PRIVATE TwoPunctures Kokkos::kokkos)
  enable_testing()
  add_test(NAME kokkos_independent COMMAND test_kokkos)
  add_executable(test_hispid_kokkos tests/test_hispid_kokkos.cpp)
  target_compile_definitions(test_hispid_kokkos PRIVATE PUNCTURES_KOKKOS=1 HISPID_ROW_POWER=${HISPID_ROW_POWER})
  target_include_directories(test_hispid_kokkos PRIVATE src include)
  target_link_libraries(test_hispid_kokkos PRIVATE HiSpID TwoPunctures Kokkos::kokkos)
  add_test(NAME hispid_kokkos_actual_kernels COMMAND test_hispid_kokkos)
  add_executable(test_hispid_setup_kokkos tests/test_hispid_setup_kokkos.cpp)
  target_compile_definitions(test_hispid_setup_kokkos PRIVATE PUNCTURES_KOKKOS=1 HISPID_ROW_POWER=${HISPID_ROW_POWER})
  target_include_directories(test_hispid_setup_kokkos PRIVATE src include)
  target_link_libraries(test_hispid_setup_kokkos PRIVATE HiSpID TwoPunctures Kokkos::kokkos)
  add_test(NAME hispid_kokkos_setup COMMAND test_hispid_setup_kokkos)
  set_tests_properties(hispid_kokkos_setup PROPERTIES SKIP_RETURN_CODE 77)
  add_executable(test_by_kokkos tests/test_by_kokkos.cpp)
  target_compile_definitions(test_by_kokkos PRIVATE PUNCTURES_KOKKOS=1)
  target_include_directories(test_by_kokkos PRIVATE src include)
  target_link_libraries(test_by_kokkos PRIVATE TwoPunctures Kokkos::kokkos)
  add_test(NAME by_kokkos_actual_kernels COMMAND test_by_kokkos)
endif()
